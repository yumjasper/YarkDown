cmake_minimum_required(VERSION 3.24)
project(YarkAcrylic LANGUAGES CXX)
add_library(YarkAcrylic SHARED plugin.cpp exports.def)
target_compile_features(YarkAcrylic PRIVATE cxx_std_17)
target_compile_definitions(YarkAcrylic PRIVATE UNICODE _UNICODE NOMINMAX)
target_compile_options(YarkAcrylic PRIVATE /utf-8 /W4)
target_link_libraries(YarkAcrylic PRIVATE d2d1 dwrite dwmapi user32 gdi32 shell32 ole32 comctl32)
set_target_properties(YarkAcrylic PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/plugins/x86-unicode")

set_property(TARGET YarkAcrylic PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
add_executable(acrylic-smoke smoke.cpp smoke.manifest)
target_compile_features(acrylic-smoke PRIVATE cxx_std_17)
target_compile_definitions(acrylic-smoke PRIVATE UNICODE _UNICODE NOMINMAX)
target_compile_options(acrylic-smoke PRIVATE /utf-8)
target_link_libraries(acrylic-smoke PRIVATE d2d1 dwrite dwmapi user32 gdi32 shell32 ole32 comctl32)
